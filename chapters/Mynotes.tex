\section{Derived functors}
\begin{itemize}
    \item(Homotopy)
    We define a equivalence relation on $\Hom (C_\bullet, D_\bullet)$ as follows: 
    Let $f,g: (C_\bullet,d)\to (D_\bullet,d)$ be two morphims of complexes, then we say that $f\sim g$ 
    if there exist $h_n:C_n\to D_{n+1},\;\forall n$ such that the diagram below commutes.
    \begin{center}
        \begin{tikzcd}
            C_{n-1}\arrow[d]\arrow[dr,swap,"h_{n-1}"]&C_n\arrow[d]\arrow[l,"d_n"]\arrow[rd,swap,"h_{n}"]
            &C_{n+1}\arrow[d]\arrow[dr,swap,"h_{n+1}"]\arrow[l,"d_{n+1}"]&\cdots\arrow[l]\\
            D_{n-1}&D_n\arrow[l,"d_n"]&D_{n+1}\arrow[l,"d_{n+1}"]&\cdots\arrow[l]
        \end{tikzcd}
    \end{center}
    We write  $h\circ d+d\circ h = g-f$. Observe that $d$ decreases the index and $h$ increases the index. So 
    $(h\circ d)_n = h_{n-1}\circ d_n$ and $(d\circ h)_n = d_{n+1}\circ h_n$. 
    \begin{definition}
        We say that $f_\bullet:C_\bullet\to D_\bullet$ is a Homotopy equivalence if there exists 
        $g_\bullet:D_\bullet\to C_\bullet$ such that $g_\bullet\circ f_\bullet \sim \text{Id}_{C_\bullet}$ and 
        $f_\bullet\circ g_\bullet\sim \text{Id}_{D_\bullet}$.
    \end{definition}
    \begin{notation}
    Normally we will avoid to write $(C_\bullet,d)$ for complexes and adote the minimalistic notation by just writing 
    ``$C$'' instead. Same with morphism of complexes.
    \end{notation}
    \begin{proposition}
        Let $f,g:C\to D, f\sim g$, then $H_n(f)=H_n(g),\forall n$. 
    \end{proposition}
    Proof: Remember that we define $H_n(f): H_n(C)\to H_n(D):[x]\mapsto [f(x)]$. But since $g-f = hd+dh$, we get that 
    $g(x)-f(x)=dh(x), x \in \text{Ker}(d: C_n \to C_{n-1})$ and $[f(x)]=[g(x)]$.\qed
    \begin{corollary}
        If $f:C\to D$ is Homotopy equivalence then $H_n(C)\cong H_n(D)$.
    \end{corollary}
    Proof: By hypothesis, there exists $g:D\to C$ such that $f \circ g\sim\text{Id}_D,g\circ f\sim\text{Id}_C$. Therefore 
    $H_n(f)\circ H_n(g) = H_n(\text{Id}_D)=\text{Id}_{H_n(D)}$ and etc.\qed 
    
    \item(Injective resolution)
    Here $\callC$ denotes an abelian category if not stated otherwise.
    \begin{definition}
        We say that $I\in\callC$ is injective if 
        $\Hom_{\callC}(-,I):\callC\to \underline{\text{Ab}}$ is an an exact functor. This is equivalent to say that 
        whenever we have a monomorphism $i: A\to B$ and a morphism $f:A\to I$, then there exists 
        $\widetilde{f}: B\to I$.
        \begin{center}
            \begin{tikzcd}
                A\arrow[d,swap,"f"]\arrow[r,"i"]&B\arrow[ld,dashed,"\exists \widetilde{f}"]\\
                I&
            \end{tikzcd}
        \end{center}
    \end{definition}
    \begin{definition}
        Let $A\in\callC$, then an injective resolution for $A$ is a complex 
        \begin{equation}
          I^\bullet = (0\to I^0\to I^1\to I^2\to\cdots)
        \end{equation}
        such that 
        \begin{itemize}
            \item[(1)] $I^n$ is injective;
            \item[(2)] The augmented complex 
            $\widetilde{I}^\bullet=(0\to A\to I^0\to I^1\to\cdots)$ is acyclic.
        \end{itemize}
     \end{definition}

     \begin{definition}
        We say that $\callC$ has enough injectives if for any $A\in\callC$ there exists an injective object $I\in\callC$ 
        and a monomorphism $f:A\to I$.
     \end{definition}
    \begin{proposition}
        If $\callC$ has enough injectives, then any $A\in\callC$ has a injective resolution.
    \end{proposition}
    Proof: Draw the diagram 
    \begin{center}
        \begin{tikzcd}
            0\arrow[r]&A\arrow[r,"\epsilon"]&I^0\arrow[rr,dashed]\arrow[rd]&&I^1\cdots\\ 
            &&&I^0/\epsilon(A)=Q^0\arrow[ru,hook]&
        \end{tikzcd}
    \end{center}
    and repeat
    \begin{center}
        \begin{tikzcd}
            I^1\arrow[rr,dashed]\arrow[rd]&&I^2\cdots\\ 
            &I^1/Q^0=Q^1\arrow[ru,hook]&
        \end{tikzcd}
    \end{center}
    etc...\qed
    \item(Derived functors)
    Assume that $\callA$ has enought injectives. If $F:\callA\to\callB$ is a left exact functor we define 
    \begin{equation}
        R^nF(A) = H^n(F(I^\bullet))
    \end{equation}
    where $I^\bullet$ is some injective resolution of $A$. 
    \begin{remark}
        We have 
        \begin{equation}
            \begin{split}
                R^0F(A) &= H^0(F(I^\bullet))\\
                &=H^0(0\to F(I^0)\to F(I^1)\to\cdots)\\
                &=\text{Ker}(F(I^0)\to F(I^1))/\text{Im}(0\to F(I^0))\\
                &=F(A)
            \end{split}
        \end{equation}
        Here we used that $F$ is left exact, so $0\to F(A)\to F(I^0)\to F(I^1)$ is exact.
    \end{remark}

    \begin{notation}
        \textcolor{red}{From now on, we will adopt the following convention: We will draw arrows aways from left to right. So, index decreases 
        from left to right in homology and increases from left to right in cohomology.}
        \begin{equation}
            \begin{split}
                &(\text{Homology})\quad\cdots\to C_2\to C_1\to C_0\to 0\\
                &(\text{Cohomology})\quad 0\to C^0\to C^1\to C^2\to \cdots
            \end{split}
        \end{equation}
    \end{notation}
    
    \begin{lemma}
        Let $f:A\to B$ be a morphism and take injective resolutions 
        $A\to I^\bullet, B\to J^\bullet$. Then there exists 
        lifting $f^\bullet:I^\bullet\to J^\bullet$,i.e., $\{f^n\}_{n\geq0}$ such that 
        \begin{center}
            \begin{tikzcd}
                0\arrow[r]&A\arrow[d,"f"]\arrow[r,"\eta"]&I^0\arrow[d,"f^0"]\arrow[r,"d"]&I^1\arrow[d,"f^1"]\arrow[r,"d"]&\cdots\\
                0\arrow[r]&B\arrow[r,"\eta"]&J^0\arrow[r,"d"]&J^1\arrow[r,"d"]&\cdots
            \end{tikzcd}
        \end{center}
    \end{lemma}
    Proof: First we define $f^0$ by the diagram below.
    \begin{center}
        \begin{tikzcd}
            A\arrow[r,"\eta"]\arrow[dr,"\eta\circ f"]&I^0\arrow[d,dashed,"\exists f^0"]\\ &J^0
        \end{tikzcd}
    \end{center}
    For the second block we construct the diagram 
    \begin{center}
        \begin{tikzcd}
            I^0\arrow[rr]\arrow[dr]\arrow[dd]&&I^1\arrow[dd]\\
            &Q^0=I^0/\text{Ker}(d^0)\arrow[ur,hook]\arrow[dd,dashed]&\\
            J^0\arrow[dr]\arrow[rr]&&J^1\\
            &R^0=J^0/\text{Ker}(d^0)\arrow[ru,hook]&
        \end{tikzcd}
    \end{center}
    The dashed morphism is defined by $Q^0\to R^0: [x]\mapsto [f^0(x)]$. It's well defined, because 
    if $x-x^\prime\in\text{Ker}(d^0)=\text{Im}(\eta)$ implies that $x-x^\prime=\eta(a)$. So 
    $f^0(x)-f^0(x^\prime)=f^0\eta(a)=\eta f(a)$ and $d\eta=0$. Now we define $f^1$ from the diagram
    \begin{center}
        \begin{tikzcd}
            &I^1\arrow[dd,dashed,"\exists f^1"]\\
            Q^0\arrow[dd]\arrow[ru,hook]\arrow[rd]&\\
            &J^1\\
            R^0\arrow[ru,hook]&
        \end{tikzcd}
    \end{center}
    and etc...\qed
    \begin{lemma}
        Any two liftings are homotopic.
    \end{lemma}
    Proof: We want to find $h$ such that makes the diagram below commutes. 
    \begin{center}
            \begin{tikzcd}
                0\arrow[r]&A\arrow[d,swap,"0"]\arrow[dl,swap,"0"]\arrow[r,"\eta"]&I^0\arrow[dl,swap,"h^0"]\arrow[d,swap,"f^0-g^0"]
                \arrow[r,"d"]&
                I^1\arrow[dl,swap,"h^1"]\arrow[d,"f^1-g^1"]\arrow[r,"d"]&\cdots\\
                0\arrow[r]&B\arrow[r,"\eta"]&J^0\arrow[r,"d"]&J^1\arrow[r,"d"]&\cdots
            \end{tikzcd}
        \end{center}
    For $n=0$, we need $h^1\circ d^0+d^1\circ h^0=g^0-f^0$. It's clear that $h^0=0$ makes the first block 
    commutes, then $h^1\circ d^0 = g^0-f^0$. However, we have 
    \begin{center}
        \begin{tikzcd}
            &Q^0=I^0/\text{Ker}(d^0)\arrow[rd,hook]\arrow[ddl]&\\
            I^0\arrow[rr]\arrow[d]\arrow[ur]&&I^1\arrow[dll,dashed]\\
            J^0&&
        \end{tikzcd}
    \end{center}
    As before, we define $Q^0\to J^0: x + \text{Ker}(d^0)\mapsto f^0(x)-g^0(x)$. Now it's well defined because if 
    $x\in\text{Ker}(d^0)=\text{Im}(\eta)$, we get $f^0(x)-g^0(x)=(f^0-g^0)\eta(a)=0$.\qed
    \begin{theorem}
        \begin{itemize}
            \item[(1)] $RF^n(A)$ independ of choice of resolution $I^\bullet$;
            \item[(2)] $f:A\to B$ induces $R^nF(A)\to R^nF(B)$;
            \item[(3)] If $0\to A\to B\to C\to 0$ is a SES, then there exists a long exact sequence:
            \begin{center}
                \begin{tikzcd}
                &\cdots&&\\
                &R^1F(A)\arrow[r]&R^1F(B)\arrow[r]&R^1F(C)\arrow[llu]\\
                0\arrow[r]&F(A)\arrow[r]&F(B)\arrow[r]&F(C)\arrow[llu]
                \end{tikzcd}
            \end{center}  
        \end{itemize}
    \end{theorem}
    Proof: Let $A\in\callC$ and take two injective resolutions $I^\bullet, J^\bullet$ of $A$. Then 
    the identity morphism $A\to A$ lifts to morphisms $f:I^\bullet \to J^\bullet, g:J^\bullet\to I^\bullet$. 
    In particular $fg: I^\bullet\to I^\bullet$ is a complex morphism, so it must be homotopic to the identity 
    morphism, i.e., $fg\sim\text{Id}_{I^\bullet}\Rightarrow F(f)\circ F(g)\sim \text{Id}_{F(I^\bullet)}$. In 
    a similar way we see that $F(g)\circ F(f)\sim \text{Id}_{F(J^\bullet)}$, so $H^n(F(I^\bullet))\cong H^n(F(J^\bullet))$. 
    The second one also follows directly from the lemmas above.\par
